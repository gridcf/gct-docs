9 /*
10  * The idea with this library is to replace OpenSSL's ERR_raise() and
11  * ERR_raise_data() with variants that are more suitable for providers that
12  * have their own error reason table.
13  *
14  * This assumes variadic function-like macros, i.e. C99 or newer.
15  *
16  * A minimal amount of preparation is needed on provider initialization and
17  * takedown:
18  *
19  * 1. The provider's outgoing OSSL_DISPATCH table must include an entry
20  * got OSSL_FUNC_PROVIDER_GET_REASON_STRINGS, with a function that returns
21  * the provider's table of reasons.
22  * That table of reasons is a simple OSSL_ITEM array, where each element
23  * contains a numeric reason identity for the reason, and the description
24  * text string for that reason.
25  * Each numeric reason identity MUST be unique within this array.
26  *
27  * 2. On provider initialization, an error handle must be created using
28  * proverr_new_handle(). The returned pointer is passed as first argument
29  * to ERR_raise() and ERR_raise_data().
30  *
31  * 3. On provider takedown, the error handle must be freed, using
32  * proverr_free_handle().
33  *
34  * With this preparation, the provider code can use ERR_raise() and
35  * ERR_raise_data() "as usual", with the exception that the first argument is
36  * the error handle instead of one of the OpenSSL ERR_LIB_ macros.
37  */
38 
39 /*
40  * In case <openssl/err.h> was included, we throw away its error recording
41  * macros.
42  * Note that ERR_put_error() is NOT recreated. It's deprecated and should not
43  * be used any more.
44  */
45 #undef ERR_put_error
46 #undef ERR_raise
47 #undef ERR_raise_data
